  export type Sex = 'male' | 'female';
  export type Goal = 'gain' | 'loss' | 'maintain';
  export type ActivityLevel = 'minimal' | 'low' | 'medium' | 'high' | 'very_high';
  export interface NutritionProfile {
    sex: Sex; ageYears: number; heightCm: number; weightKg: number;
    activityLevel: ActivityLevel; goal: Goal;
    desiredRateKgPerMonth?: number;   // optional; clamped to [0, MAX_RATE_KG_PER_MONTH]
  }
  export interface NutritionTargets {
    bmr: number; tdee: number; targetKcal: number;
    proteinG: number; fatG: number; carbsG: number;
    floorApplied: boolean;            // true when the safety floor overrode the deficit
    rateKgPerMonth: number;           // the rate actually used after clamping
  }

  calculateTargetMacros(weightKg, targetKcal): { proteinG, fatG, carbsG }
  calculateNutritionTargets(profile: NutritionProfile): NutritionTargets

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Onboarding input (Phase 2, untrusted user text) -> domain functions | Age/height/weight/rate arrive as user-typed numbers |
| Domain output -> user-facing health guidance | Wrong numbers are the product's core failure mode |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-14 | Tampering (input) | `calculateBmr`, `calculateTdee`, `calculateTargetCalories`, `calculateTargetMacros` | mitigate | Every public function validates that its numeric arguments are finite and positive and throws a field-named Error; `NaN`/negative/zero inputs can never produce a plausible-looking target |
| T-01-15 | Tampering (constraint bypass) | `desiredRateKgPerMonth` parameter | mitigate | Rate is clamped to `MAX_RATE_KG_PER_MONTH = 1` inside the domain function, so no caller (including a future API or a buggy Phase 2 keyboard) can request a faster deficit; the `users` table check constraint from Plan 03 is the second layer |
| T-01-16 | Denial of service (user health) | Safety calorie floor | mitigate | `CALORIE_FLOOR_MALE`/`CALORIE_FLOOR_FEMALE` are hard-coded per TECH_SPEC §6.3, applied after the rate adjustment, and `floorApplied` is returned so the UI can explain the override rather than hide it |
| T-01-17 | Information Disclosure | Domain layer | accept | These functions handle health data in-memory only, perform no I/O and no logging — nothing to disclose at this layer |
