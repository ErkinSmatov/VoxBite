  export const CANDIDATE_COUNT = 3;                                  // MATCH-01
  export const ALLOWED_SOURCES = ['foundation_food', 'sr_legacy_food'] as const;  // MATCH-02
  export interface FdcCandidate {
    fdcId: number;
    description: string;                       // shown verbatim to the user — states raw/cooked/method
    source: (typeof ALLOWED_SOURCES)[number];
    kcal: number | null;
    proteinG: number | null;
    fatG: number | null;
    carbsG: number | null;
    sugarG: number | null;                     // null = "нет данных" (TECH_SPEC §5.8), never 0
    similarity: number;                        // 1 - cosine distance, higher is closer
  }
  export interface FdcRepository {
    findNearest(embedding: number[], limit: number): Promise<FdcCandidate[]>;
  }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| `fdc_foods` contents -> user-facing candidate list | Whatever is in the table is shown to the user as authoritative nutrition data |
| Ingredient text (Phase 3: LLM output, ultimately user speech) -> embedding -> vector query | Untrusted text reaches the query path |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-27 | Tampering (MATCH-02 at read time) | `matchIngredient` | mitigate | `ALLOWED_SOURCES` allow-list filter applied to every repository result before truncation, with over-fetch so filtering never shortens the list; unit-tested with a `branded_food` row; independently asserted again by `verify-matches.ts` |
| T-01-28 | Tampering (silent nutrient falsification) | `fdc-repository.ts` mapping | mitigate | Nullable nutrients pass through untouched; a static check rejects `?? 0` in the repository; the domain function performs no nutrient math at all |
| T-01-29 | Injection | Vector query built from untrusted ingredient text | mitigate | The text never reaches SQL — only its 1536-number embedding does, passed as a bound parameter through Drizzle's `cosineDistance()`; no string interpolation into the query |
| T-01-30 | Denial of Service (performance) | `findNearest` | mitigate | Search is executed by Postgres against the HNSW `vector_cosine_ops` index with a `LIMIT`, never by loading rows into Node; static check forbids a JS distance loop |
| T-01-03 | Denial of Service (owner's budget) | `verify-matches.ts` embedding call | accept | One batched call for 10 short strings per invocation — cost is below $0.000001 and is printed by the script |
</threat_model>
